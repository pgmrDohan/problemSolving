## 2. 코드 설계 (Pseudo Code)

```cpp
#include <iostream>
#include <vector>
using namespace std;

int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int N = 0;
  int M = 0;
  cin >> N >> M;
  vector<int> v(N);
  for (int i = 0; i < N; i++) {
    cin >> v[i];
  }
  for (int i = 0; i < M; i++) {
    int foo = 0, bar = 0;
    cin >> foo >> bar;
    foo--;
    bar--;
    int sum = 0;
    for (int j = foo; j <= bar; j++) {
      sum += v[j];
    }
    cout << sum << '\n';
  }
  return 0;
}
```

## 5. 최종 코드

```cpp
#include <iostream>
#include <vector>
using namespace std;

int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int N = 0;
  int M = 0;
  cin >> N >> M;
  vector<int> v(N);
  vector<int> sums(N);
  for (int i = 0; i < N; i++) {
    cin >> v[i];
    sums[i] = v[i] + sums[(i - 1) < 0 ? 0 : (i - 1)];
  }
  for (int i = 0; i < M; i++) {
    int foo = 0, bar = 0;
    cin >> foo >> bar;
    foo--;
    bar--;
    cout << sums[bar] - sums[foo - 1] << '\n';
  }
  return 0;
}
```
