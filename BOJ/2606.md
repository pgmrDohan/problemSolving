#include <iostream>
#include <stack>
#include <utility>
#include <vector>
using namespace std;

int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int numsOfComs = 0, numsOfConnectionPair = 0;
  cin >> numsOfComs >> numsOfConnectionPair;
  vector<pair<int, int>> connectionVec(numsOfConnectionPair + 1);
  for (int i = 1; i <= numsOfConnectionPair; i++) {
    cin >> connectionVec[i].first >> connectionVec[i].second;
  }
  /*    인접 리스트
    vector<vector<int>> adj(numsOfComs + 1);
    for (int i = 0; i < numsOfConnectionPair; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
  */
  int visitCount = 0;
  stack<int> visitWait;
  vector<bool> visited(numsOfConnectionPair + 1);
  visited.insert(visited.begin(), numsOfConnectionPair + 1, 0);
  visitWait.push(1);
  while (!visitWait.empty()) {
    int vTop = visitWait.top();
    visitWait.pop();
    visited[vTop] = 1;
    visitCount++;
    for (int i = 1; i <= numsOfConnectionPair; i++) {
      if (connectionVec[i].first == vTop) {
        if (visited[connectionVec[i].second])
          continue;
        else
          visited[connectionVec[i].second] = 1;
        visitWait.push(connectionVec[i].second);
      } else if (connectionVec[i].second == vTop) {
        if (visited[connectionVec[i].first])
          continue;
        else
          visited[connectionVec[i].first] = 1;
        visitWait.push(connectionVec[i].first);
      }
    }
    /*  인접 리스트 활용을 통한 간소화
        for (int next : adj[curr]) {
            if (!visited[next]) {
                visited[next] = true;
                st.push(next);
            }
        }
    */
  }
  cout << visitCount - 1 << '\n';
  return 0;
}
