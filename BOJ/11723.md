## 5. 최종 코드

```cpp
#include <array>
#include <iostream>
#include <string>
using namespace std;

int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int N = 0;
  cin >> N;
  array<int, 20> v{};
  int idx = 0;
  for (int i = 0; i < N; i++) {
    string cmd;
    cin >> cmd;
    int val = 0;
    if (cmd == "add") {
      cin >> val;
      v[val - 1] = 1;
    }
    if (cmd == "remove") {
      cin >> val;
      v[val - 1] = 0;
    }

    if (cmd == "check") {
      cin >> val;
      cout << (v[val - 1] ? 1 : 0) << '\n';
    }
    if (cmd == "toggle") {
      cin >> val;
      v[val - 1] = !v[val - 1];
    }
    if (cmd == "all") {
      for (int i = 0; i < 20; i++) {
        v[i] = 1;
      }
    }
    if (cmd == "empty") {
      for (int i = 0; i < 20; i++) {
        v[i] = 0;
      }
    }
  }
  return 0;
}
```
