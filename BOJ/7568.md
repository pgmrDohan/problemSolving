## 5. 최종 코드

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <utility>
using namespace std;

int main (void) {
    int N = 0;
    cin >> N;
    vector<pair<int,int>> stds(N);
    for (int i = 0; i < N; i++) {
        cin >> stds[i].first >> stds[i].second;
    }
    for (int i = 0; i < N; i++) {
        int k = 0;
        for (int j = 0; j < N; j++) {
            if(j==i) continue;
            if(stds[j].first > stds[i].first && stds[j].second > stds[i].second) k++;
        }
        cout << k + 1 << '\n';
    }
}
```
