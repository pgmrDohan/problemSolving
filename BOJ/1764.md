## 5. 최종 코드

```cpp
#include <algorithm>
#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int N = 0, M = 0;
  cin >> N >> M;
  vector<string> nr(N);
  vector<string> ns(M);
  vector<int> nrs;
  for (int i = 0; i < N; i++) {
    cin >> nr[i];
  }
  for (int j = 0; j < M; j++) {
    cin >> ns[j];
  }
  sort(nr.begin(), nr.end());
  sort(ns.begin(), ns.end());
  for (int i = 0; i < N; i++) {
    int s = 0;
    int e = M - 1;
    while (s <= e) {
      if (nr[i] == ns[(e + s) / 2]) {
        nrs.push_back(i);
        break;
      } else {
        if (nr[i] > ns[(e + s) / 2]) {
          s = (e + s) / 2 + 1;
        } else {
          e = (e + s) / 2 - 1;
        }
      }
    }
  }
  cout << nrs.size() << '\n';
  for (int i = 0; i < nrs.size(); i++) {
    cout << nr[nrs[i]] << '\n';
  }
  return 0;
}
```

## 6. 메모

- 투포인터가 더 좋은 알고리즘. 아래와 같이 풀수 있음. (답에서 가져옴.)

```cpp
#include <algorithm>
#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);

  int N, M;
  cin >> N >> M;

  vector<string> nr(N), ns(M);

  for (int i = 0; i < N; i++) cin >> nr[i];
  for (int i = 0; i < M; i++) cin >> ns[i];

  sort(nr.begin(), nr.end());
  sort(ns.begin(), ns.end());

  vector<string> result;

  int i = 0, j = 0;

  while (i < N && j < M) {
    if (nr[i] == ns[j]) {
      result.push_back(nr[i]);
      i++;
      j++;
    } else if (nr[i] < ns[j]) {
      i++;
    } else {
      j++;
    }
  }

  cout << result.size() << '\n';
  for (auto &name : result) {
    cout << name << '\n';
  }

  return 0;
}
```
