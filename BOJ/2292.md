## 5. 최종 코드

```cpp
#include <iostream>
using namespace std;

int main(void) {
  int n = 0;
  cin >> n;
  int i = 0;
  int cod = 1 + 6 * i;
  while (1) {
    if (i < n && n <= cod) {
      break;
    }
    i++;
    cod += 6 * i;
  }
  cout << i + 1 << '\n';
}
```
