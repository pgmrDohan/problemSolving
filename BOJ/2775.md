## 5. 최종 코드

```cpp
#include <iostream>
using namespace std;

int c (int a, int b) {
    if(a == 0) return b;
    int sum = 0;
    for(int i = 1; i<=b; i++) {
        sum += c(a-1,i);
    }
    return sum;
}

int main (void) {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int N = 0;
    cin >> N;
    for (int i = 0; i<N; i++) {
        int a = 0, b = 0;
        cin >> a >> b;
        cout << c(a,b) << '\n';
    }
    return 0;
}
```
