## 5. 최종 코드

```cpp
#include <array>
#include <iostream>
using namespace std;

int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  array<int, 12> dp;
  dp[1] = 1;
  dp[2] = 2;
  dp[3] = 4;
  int T = 0;
  int calced = 3;
  cin >> T;
  for (int i = 0; i < T; i++) {
    int target = 0;
    cin >> target;
    if (calced < target)
      for (int j = 4; j <= target; j++) {
        dp[j] = dp[j - 1] + dp[j - 2] + dp[j - 3];
      }
    cout << dp[target] << '\n';
  }

  return 0;
}
```
