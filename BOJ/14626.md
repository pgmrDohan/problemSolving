## 5. 최종 코드

```cpp
#include <iostream>
#include <string>
using namespace std;

int main (void) {
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
    string target = "";
    int ans = 0;
    cin >> target;
    int m = target[12]-'0';
    int sum = 0;
    int mul = 0;
    for(int i = 0; i<12; i++){
        if(target[i]=='*') {mul = ((i%2==0)?1:3); continue;}
        sum += ((i%2==0)?1:3)*(int)(target[i]-'0');
    }
    for(int i = 1; i<10; i++) {
        if(m==0) {
            if ((sum+mul*i) % 10==0) {ans = i; break;}
        } else {
            if (10-m == (sum+mul*i) % 10) {ans = i; break;}
        }
    }
    cout << ans << '\n';
    return 0;
}
```
