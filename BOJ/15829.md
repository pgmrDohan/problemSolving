## 5. 최종 코드

```cpp
#include <iostream>
#include <cmath>
using namespace std;

int main (void) {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int N = 0;
    cin >> N;
    long long sum = 0;
    long long p = 1;
    for(int i = 0; i < N; i++) {
        char foo = 'a';
        cin >> foo;
        sum += ((int)foo - 96)*p;
        p*=31;
        p %= 1234567891;
        sum %= 1234567891;
    }
    cout << sum << '\n';
}
```
