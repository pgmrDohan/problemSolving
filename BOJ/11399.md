## 2. 코드 설계 (Pseudo Code)

```cpp
#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int N = 0;
  cin >> N;
  vector<int> v(N);
  for (int i = 0; i < N; i++) {
    cin >> v[i];
  }
  int min = 1000000;
  do {
    int t = 0;
    int foo = 0;
    for (int i = 0; i < N; i++) {
      foo += v[i];
      t += foo;
    }
    if (t < min) {
      min = t;
    }
  } while (next_permutation(v.begin(), v.end()));
  cout << min << '\n';
}
```
