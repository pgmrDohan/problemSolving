## 5. 최종 코드

```cpp
#include <iostream>
#include <string>
#include <array>
using namespace std;

int main (void) {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    array<string,3> target;
    cin >> target[0] >> target[1] >> target[2];
    int m_traget = 0;
    for (int i = 0; i < 3; i++) {
        if (target[i]!="Fizz" && target[i]!="Buzz" && target[i]!="FizzBuzz") {
            m_traget = stoi(target[i])+3-i;
            break;
        }
    }
    if(m_traget % 3 == 0 && m_traget % 5 == 0) {
        cout << "FizzBuzz" << '\n';
    } else if(m_traget % 3 == 0 && m_traget % 5 != 0) {
        cout << "Fizz" << '\n';
    } else if(m_traget % 3 != 0 && m_traget % 5 == 0) {
        cout << "Buzz" << '\n';
    } else {
        cout << m_traget << '\n';
    }
    return 0;
}
```
