## 5. 최종 코드

```cpp
#include <iostream>
#include <string>
using namespace std;

int main (void) {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int N = 0;
    cin >> N;
    if(N == 1) {cout << 0 << '\n'; return 0;}
    for(int i = 1; i<=N*2; i++) {
        string test = to_string(i);
        int sum = 0;
        for(int j = 0; j<test.size(); j++) {
            sum += (int)(test[j] - '0');
        }
        if(i+sum == N) {
            cout << i << '\n';
            return 0;
        }
    }
    cout << 0 << '\n';
    return 0;
}
```
