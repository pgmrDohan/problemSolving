## 2. 코드 설계 (Pseudo Code)

```cpp
#include <iostream>
using namespace std;

int fibonacci(int n, int* zero, int* one) {
  if (n == 0) {
    (*zero)++;
    return 0;
  } else if (n == 1) {
    (*one)++;
    return 1;
  } else {
    return fibonacci(n - 1, zero, one) + fibonacci(n - 2, zero, one);
  }
}

int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int T = 0;
  cin >> T;
  for (int i = 0; i < T; i++) {
    int foo = 0;
    cin >> foo;
    int zero = 0, one = 0;
    fibonacci(foo, &zero, &one);
    cout << zero << ' ' << one << '\n';
  }
  return 0;
}
```

  array<pair<int, int>, 41> fibonacci;
  fibonacci[0] = {1, 0};
  fibonacci[1] = {0, 1};
  int computed = 1;
  for (int i = 0; i < T; i++) {
    int foo = 0;
    cin >> foo;
    for (int j = computed + 1; j <= foo; j++) {
      fibonacci[j] = {(fibonacci[j - 1].first + fibonacci[j - 2].first),
                      (fibonacci[j - 1].second + fibonacci[j - 2].second)};
    }
    cout << fibonacci[foo].first << ' ' << fibonacci[foo].second << '\n';
    if (foo > computed) computed = foo;
  }
